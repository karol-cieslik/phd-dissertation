% The template is intentionally based on the standard report class so it is
% straightforward to adapt. Check your faculty's current submission rules
% before using the layout for a final deposit.

\documentclass[12pt,twoside]{report}

% Page layout: extra binding space on the inner edge for duplex printing.
\usepackage[a4paper,inner=35mm,outer=25mm,top=25mm,bottom=25mm,headheight=14pt]{geometry}
\usepackage{indentfirst}
\setlength{\parindent}{0.75cm}
\linespread{1.15}

% Language and typography. The last babel language is the default.
\usepackage[polish,english]{babel}
\usepackage[T1]{fontenc}
\usepackage{microtype}

% Mathematics, chemistry, and physical units.
\usepackage{amsmath}
\usepackage[version=3]{mhchem}

\DeclareSIUnit{\ions}{ions}

% Figures, tables, and rotated material.
\usepackage{graphicx}
\usepackage{rotating}
\usepackage[font=small,labelfont=bf]{caption}
\usepackage{booktabs}
\usepackage[flushleft]{threeparttable}

% Page furniture and restrained heading styling.
\usepackage{fancyhdr}
\usepackage{titlesec}
\usepackage{xcolor}
\definecolor{ThesisInk}{HTML}{24364B}
\titleformat{\chapter}[display]
  {\normalfont\bfseries\color{ThesisInk}}
  {\filleft\Large\chaptertitlename\ \thechapter}
  {1ex}
  {\titlerule\vspace{1ex}\filright\Huge}
\titleformat{name=\chapter,numberless}[display]
  {\normalfont\bfseries\color{ThesisInk}}
  {}
  {0pt}
  {\titlerule\vspace{1ex}\LARGE}
\titlespacing*{\chapter}{0pt}{-12pt}{24pt}
\titleformat{\section}
  {\normalfont\Large\bfseries\color{ThesisInk}}
  {\thesection}{1em}{}
\titleformat{\subsection}
  {\normalfont\large\bfseries\color{ThesisInk}}
  {\thesubsection}{1em}{}

\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\thepage}
\fancyhead[LO]{\small\scshape\nouppercase{\rightmark}}
\fancyhead[RE]{\small\scshape\nouppercase{\leftmark}}
\renewcommand{\headrulewidth}{0.4pt}
\fancypagestyle{plain}{%
  \fancyhf{}
  \fancyfoot[C]{\thepage}
  \renewcommand{\headrulewidth}{0pt}}
\setlist[itemize]{topsep=4pt,itemsep=2pt}
\setlength{\belowcaptionskip}{3pt}

% Small helpers used by the examples below.
\newcommand{\horrule}[1]{\rule{\linewidth}{#1}}

\newcommand{\ra}[1]{\renewcommand{\arraystretch}{#1}}

% Load hyperref after packages that affect headings, captions, or links.
\usepackage[unicode=true,pdfusetitle]{hyperref}
\hypersetup{
  colorlinks=true,
  linkcolor=ThesisInk,
  citecolor=ThesisInk,
  urlcolor=ThesisInk
}
